{{/* Input: {image, strokeOverride?}. Returns image with stroke and border applied based on site.Params.gallery. strokeOverride bypasses scaling and applies that many pixels directly — use for full-size images. */}} {{ $image := .image }} {{ $origWidth := $image.Width }} {{ $fullSize := site.Params.gallery.fullSize | default 1600 | int }} {{ $strokeSize := site.Params.gallery.strokeSize | default 0 | int }} {{ if gt $strokeSize 0 }} {{ $stroke := div (mul $strokeSize $origWidth) $fullSize }} {{ with .strokeOverride }}{{ if gt . 0 }}{{ $stroke = . }}{{ end }}{{ end }} {{ $image = $image.Filter (images.Padding $stroke (site.Params.gallery.strokeColor | default "#000000")) }} {{ end }} {{ $borderSize := site.Params.gallery.borderSize | default 0 | int }} {{ if gt $borderSize 0 }} {{ $border := div (mul $borderSize $origWidth) $fullSize }} {{ $image = $image.Filter (images.Padding $border (site.Params.gallery.borderColor | default "#ffffff")) }} {{ end }} {{ return $image }}
